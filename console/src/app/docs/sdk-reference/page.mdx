# SDK Reference

Complete method reference for the TypeScript and Python SDKs.

## Sandbox Lifecycle

<MethodCard
  name="Sandbox.create"
  signature="static create(request: CreateSandboxRequest, auth?: SandboxAuth): Promise<Sandbox>"
  description="Create a new sandbox with the specified runtime and configuration. Blocks until the sandbox reaches the ready state."
>
  <ParamTable params={[
    { name: "request", type: "CreateSandboxRequest", required: true, description: "Runtime, image, resources, and other sandbox configuration." },
    { name: "request.env", type: "Record<string, string>", required: false, description: "Container environment variables for the sandbox lifetime. See [Environment guide](/docs/guides/environment)." },
    { name: "auth", type: "SandboxAuth", required: false, description: "Override default auth (controlUrl, token, tenant). Defaults from SANDBOX_* env vars on the SDK host." },
  ]} />
  <CodeTabs
    ts={`import { Sandbox } from "@sandbox-platform/sdk";

const sandbox = await Sandbox.create({
  runtime: "node24",
  image: "sandbox-runtime:node24-latest",
  resources: { vcpus: 2 },
  timeout_ms: 600_000,
  env: { NODE_ENV: "production", LOG_LEVEL: "info" },
});`}
    python={`from sandbox_platform import Sandbox, CreateSandboxRequest, ResourcesSpec

sandbox = Sandbox.create(CreateSandboxRequest(
    runtime="node24",
    image="sandbox-runtime:node24-latest",
    resources=ResourcesSpec(vcpus=2),
    timeout_ms=600_000,
    env={"NODE_ENV": "production", "LOG_LEVEL": "info"},
))`}
  />
</MethodCard>

## Command Execution

<MethodCard
  name="sandbox.runCommand"
  signature="runCommand(cmd: string, options?: RunCommandOptions): Promise<RunCommandResult>"
  description="Run a command synchronously in the sandbox and return stdout, stderr, and exit code."
>
  <ParamTable params={[
    { name: "cmd", type: "string", required: true, description: "The executable to run (e.g. '/bin/bash')." },
    { name: "options.args", type: "string[]", required: false, description: "Command arguments." },
    { name: "options.env", type: "Record<string, string>", required: false, description: "Environment variables for this command." },
    { name: "options.cwd", type: "string", required: false, description: "Working directory." },
    { name: "options.timeout_ms", type: "number", required: false, description: "Per-command timeout." },
    { name: "options.user", type: "string", required: false, description: "Run as this user." },
  ]} />
  <CodeTabs
    ts={`const result = await sandbox.runCommand("/bin/bash", {
  args: ["-c", "echo $LOG_LEVEL"],
  env: { LOG_LEVEL: "debug" },
});
console.log(result.stdout);
console.log(result.exitCode);`}
    python={`result = sandbox.run_command(
    "/bin/bash",
    args=["-c", "echo $LOG_LEVEL"],
    env={"LOG_LEVEL": "debug"},
)
print(result["stdout"])
print(result["exit_code"])`}
  />
</MethodCard>

<MethodCard
  name="sandbox.runCommand (detached)"
  signature="runCommand(cmd: string, options: { detached: true, ... }): Promise<SpawnResult>"
  description="Start a long-running process in the background. Returns a command ID for polling."
>
  <ParamTable params={[
    { name: "cmd", type: "string", required: true, description: "The executable to run." },
    { name: "options.detached", type: "true", required: true, description: "Must be true for detached mode." },
    { name: "options.args", type: "string[]", required: false, description: "Command arguments." },
    { name: "options.pty", type: "boolean", required: false, description: "Allocate a PTY for the process." },
  ]} />
  <CodeTabs
    ts={`const { cmdId } = await sandbox.runCommand("node", {
  args: ["server.js"],
  detached: true,
});
// Poll status later:
const status = await sandbox.getProcessStatus(cmdId);`}
    python={`result = sandbox.run_command_detached("/usr/bin/node", args=["server.js"])
cmd_id = result["cmd_id"]
# Poll status later:
status = sandbox.get_process_status(cmd_id)`}
  />
</MethodCard>

## Networking

<MethodCard
  name="sandbox.listPorts"
  signature="listPorts(): Promise<ListeningPort[]>"
  description="List all ports with active listeners inside the sandbox."
>
  <CodeTabs
    ts={`const ports = await sandbox.listPorts();
for (const p of ports) {
  console.log(p.port, p.protocol, p.process_name);
}`}
    python={`ports = sandbox.list_ports()
for p in ports:
    print(p["port"], p["protocol"], p.get("process_name"))`}
  />
</MethodCard>

<MethodCard
  name="sandbox.waitForPort"
  signature="waitForPort(port: number, timeoutMs?: number): Promise<void>"
  description="Block until a port has an active listener, or timeout."
>
  <ParamTable params={[
    { name: "port", type: "number", required: true, description: "Port number to wait for." },
    { name: "timeoutMs", type: "number", required: false, description: "Max wait time in milliseconds (default: 30000)." },
  ]} />
  <CodeTabs
    ts={`// Start a server, then wait for it
await sandbox.runCommand("node", { args: ["server.js"], detached: true });
await sandbox.waitForPort(3000, 10_000);`}
    python={`sandbox.run_command_detached("node", args=["server.js"])
# waitForPort is TS-only; poll list_ports in Python:
import time
for _ in range(30):
    ports = sandbox.list_ports()
    if any(p["port"] == 3000 for p in ports):
        break
    time.sleep(1)`}
  />
</MethodCard>

<MethodCard
  name="sandbox.setNetworkPolicy"
  signature="setNetworkPolicy(policy: NetworkPolicy): Promise<NetworkPolicy>"
  description="Update the sandbox's egress network policy."
>
  <ParamTable params={[
    { name: "policy.mode", type: "'allow-all' | 'deny-all' | 'rules'", required: true, description: "Default egress posture for the sandbox." },
    { name: "policy.rules", type: "NetworkRule[]", required: false, description: "Required when mode is rules. Match by domains, cidrs, ports, and allow/deny action." },
  ]} />
  <CodeTabs
    ts={`await sandbox.setNetworkPolicy({
  mode: "rules",
  rules: [
    { domains: ["api.openai.com"], ports: [443], action: "allow" },
    { domains: ["*.github.com"], ports: [443], action: "allow" },
  ],
});`}
    python={`from sandbox_platform import NetworkPolicy, NetworkRule

sandbox.set_network_policy(NetworkPolicy(
    mode="rules",
    rules=[
        NetworkRule(domains=["api.openai.com"], ports=[443], action="allow"),
        NetworkRule(domains=["*.github.com"], ports=[443], action="allow"),
    ],
))`}
  />
</MethodCard>
